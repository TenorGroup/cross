if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  cmake_minimum_required(VERSION 3.16)
  project(ReaderTapZonesTests LANGUAGES CXX)
  enable_testing()
endif()
add_executable(reader_tap_zones zones.cpp)
target_compile_features(reader_tap_zones PRIVATE cxx_std_17)
target_compile_options(reader_tap_zones PRIVATE -UNDEBUG -Wall -Wextra -Werror)
add_test(NAME ReaderTapZones.zones COMMAND reader_tap_zones)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME ReaderTapZones.bands COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_bands.py")
add_executable(reader_tap_tip tip.cpp)
target_compile_features(reader_tap_tip PRIVATE cxx_std_17)
target_compile_options(reader_tap_tip PRIVATE -UNDEBUG -Wall -Wextra -Werror)
add_test(NAME ReaderTapZones.tip COMMAND reader_tap_tip)
add_test(NAME ReaderTapZones.tip_input COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_tip_input.py")
add_test(NAME ReaderTapZones.auto_turn_stop COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_auto_turn_stop.py")
add_executable(reader_foot_band foot_band.cpp)
target_compile_features(reader_foot_band PRIVATE cxx_std_17)
target_compile_options(reader_foot_band PRIVATE -UNDEBUG -Wall -Wextra -Werror)
target_include_directories(reader_foot_band PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../../src)
add_test(NAME ReaderTapZones.foot_band COMMAND reader_foot_band)
