set(SCRIBBLE_BOUNDARY_STANDALONE OFF)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  set(SCRIBBLE_BOUNDARY_STANDALONE ON)
  cmake_minimum_required(VERSION 3.16)
  project(ScribbleBoundaryTest LANGUAGES CXX)
  enable_testing()
  get_filename_component(REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
endif()
set(CMAKE_CXX_STANDARD 20)
set(BUILD_GMOCK OFF CACHE BOOL "" FORCE)
set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
include(GoogleTest)
if(SCRIBBLE_BOUNDARY_STANDALONE)
  if(NOT DEFINED GTEST_SOURCE OR GTEST_SOURCE STREQUAL "")
    message(FATAL_ERROR "Standalone ScribbleBoundaryTest requires -DGTEST_SOURCE=/path/to/googletest")
  endif()
  add_subdirectory("${GTEST_SOURCE}" googletest-build EXCLUDE_FROM_ALL)
endif()

file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/corpus")
file(COPY "${REPO_ROOT}/test/scribble/samples.txt"
          "${REPO_ROOT}/test/scribble/samples-x4pro-1.txt"
     DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/corpus")
configure_file(samples-x4pro-loose.txt "${CMAKE_CURRENT_BINARY_DIR}/corpus/samples-x4pro-loose.txt" COPYONLY)

add_executable(ScribbleCorpusTest "${REPO_ROOT}/test/scribble/ScribbleTest.cpp")
target_link_libraries(ScribbleCorpusTest PRIVATE GTest::gtest_main)
target_include_directories(ScribbleCorpusTest PRIVATE "${REPO_ROOT}/lib")
target_compile_options(ScribbleCorpusTest PRIVATE -Wno-error=character-conversion)
target_compile_definitions(ScribbleCorpusTest PRIVATE SCRIBBLE_SAMPLES_DIR="${CMAKE_CURRENT_BINARY_DIR}/corpus")
gtest_discover_tests(ScribbleCorpusTest)

add_executable(ScribbleBoundaryTest boundary.cpp)
target_compile_features(ScribbleBoundaryTest PRIVATE cxx_std_17)
target_include_directories(ScribbleBoundaryTest PRIVATE "${REPO_ROOT}/lib")
target_compile_options(ScribbleBoundaryTest PRIVATE -Wall -Wextra -Werror)
add_test(NAME ScribbleBoundary COMMAND ScribbleBoundaryTest)
